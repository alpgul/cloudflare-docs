```js
export default {
	async fetch(request) {
		const response = await fetch("https://example.com");

		// Yanıtı klonlayın, böylece artık değişmez olmaz
		const newResponse = new Response(response.body, response);

		// Bir değerle özel başlık ekleyin
		newResponse.headers.append(
			"x-workers-hello",
			"Hello from Cloudflare Workers",
		);

		// Başlıkları silin
		newResponse.headers.delete("x-header-to-delete");
		newResponse.headers.delete("x-header2-to-delete");

		// Mevcut bir başlık için değeri ayarlayın
		newResponse.headers.set("x-header-to-change", "NewValue");

		return newResponse;
	},
};
```

```py
from workers import Response, fetch, WorkerEntrypoint

class Default(WorkerEntrypoint):
  async def fetch(self, request):
      response = await fetch("https://example.com")

      # Değiştirilebilmesi için yanıt başlıklarını alın
      new_headers = response.headers

      # Bir değerle özel başlık ekleyin
      new_headers["x-workers-hello"] = "Hello from Cloudflare Workers"

      # Başlıkları silin
      if "x-header-to-delete" in new_headers:
          del new_headers["x-header-to-delete"]
      if "x-header2-to-delete" in new_headers:
          del new_headers["x-header2-to-delete"]

      # Mevcut bir başlık için değeri ayarlayın
      new_headers["x-header-to-change"] = "NewValue"

      return Response(response.body, headers=new_headers)
```

</TabItem> <TabItem label="Hono" icon="seti:typescript">

```ts
import { Hono } from 'hono';

const app = new Hono();

app.use('*', async (c, next) => {
  // İsteği bir sonraki middleware/handler ile işleyin
  await next();

  // Yanıt oluşturulduktan sonra, başlıklarını değiştirebiliriz

  // Bir değerle özel başlık ekleyin
  c.res.headers.append(
    "x-workers-hello",
    "Hello from Cloudflare Workers with Hono"
  );

  // Başlıkları silin
  c.res.headers.delete("x-header-to-delete");
  c.res.headers.delete("x-header2-to-delete");

  // Mevcut bir başlık için değeri ayarlayın
  c.res.headers.set("x-header-to-change", "NewValue");
});

app.get('*', async (c) => {
  // example.com'dan içerik getirin
  const response = await fetch("https://example.com");

  // Orijinal başlıklarla yanıt gövdesini döndürün
  // (middleware'imiz göndermeden önce başlıkları değiştirecek)
  return new Response(response.body, {
    headers: response.headers
  });
});

export default app;
```
