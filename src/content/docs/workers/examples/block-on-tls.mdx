```js
export default {
	async fetch(request) {
		try {
			const tlsVersion = request.cf.tlsVersion;
			// Yalnızca TLS sürümleri 1.2 ve 1.3'e izin ver
			if (tlsVersion !== "TLSv1.2" && tlsVersion !== "TLSv1.3") {
				return new Response("Lütfen TLS sürümü 1.2 veya daha yüksek kullanın.", {
					status: 403,
				});
			}
			return fetch(request);
		} catch (err) {
			console.error(
				"request.cf önizleyicide mevcut değil, yalnızca üretimde",
			);
			return new Response(`Workers scriptinde hata ${err.message}`, {
				status: 500,
			});
		}
	},
};
```

```ts
import { Hono } from "hono";

const app = new Hono();

// TLS sürümünü kontrol etmek için middleware
app.use("*", async (c, next) => {
	// TLS bilgisi ile cf nesnesini almak için ham isteğe erişin
	const request = c.req.raw;
	const tlsVersion = request.cf?.tlsVersion;

	// Yalnızca TLS sürümleri 1.2 ve 1.3'e izin ver
	if (tlsVersion !== "TLSv1.2" && tlsVersion !== "TLSv1.3") {
		return c.text("Lütfen TLS sürümü 1.2 veya daha yüksek kullanın.", 403);
	}

	await next();

});

app.onError((err, c) => {
		console.error(
			"request.cf önizleyicide mevcut değil, yalnızca üretimde",
		);
		return c.text(`Workers scriptinde hata: ${err.message}`, 500);
});

app.get("/", async (c) => {
	return c.text(`TLS Sürümü: ${c.req.raw.cf.tlsVersion}`);
});

export default app;
```

```py
from workers import WorkerEntrypoint, Response, fetch

class Default(WorkerEntrypoint):
    async def fetch(self, request):
        tls_version = request.cf.tlsVersion
        if tls_version not in ("TLSv1.2", "TLSv1.3"):
            return Response("Lütfen TLS sürümü 1.2 veya daha yüksek kullanın.", status=403)
        return fetch(request)
```
