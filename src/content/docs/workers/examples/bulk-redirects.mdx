export default {
	async fetch(request) {
		const externalHostname = "examples.cloudflareworkers.com";

		const redirectMap = new Map([
			["/bulk1", "https://" + externalHostname + "/redirect2"],
			["/bulk2", "https://" + externalHostname + "/redirect3"],
			["/bulk3", "https://" + externalHostname + "/redirect4"],
			["/bulk4", "https://google.com"],
		]);

		const requestURL = new URL(request.url);
		const path = requestURL.pathname;
		const location = redirectMap.get(path);

		if (location) {
			return Response.redirect(location, 301);
		}
		// Eğer istek haritada yoksa, orijinal isteği döndürün
		return fetch(request);
	},
};

```py
from workers import WorkerEntrypoint, Response, fetch
from urllib.parse import urlparse

class Default(WorkerEntrypoint):
    async def fetch(self, request):
        external_hostname = "examples.cloudflareworkers.com"

        redirect_map = {
          "/bulk1": "https://" + external_hostname + "/redirect2",
          "/bulk2": "https://" + external_hostname + "/redirect3",
          "/bulk3": "https://" + external_hostname + "/redirect4",
          "/bulk4": "https://google.com",
          }

        url = urlparse(request.url)
        location = redirect_map.get(url.path, None)

        if location:
            return Response.redirect(location, 301)

        # Eğer istek haritada yoksa, orijinal isteği döndürün
        return fetch(request)
```

```ts
import { Hono } from "hono";

const app = new Hono();

// Yönlendirmelerinizi yapılandırın
const externalHostname = "examples.cloudflareworkers.com";

const redirectMap = new Map([
	["/bulk1", `https://${externalHostname}/redirect2`],
	["/bulk2", `https://${externalHostname}/redirect3`],
	["/bulk3", `https://${externalHostname}/redirect4`],
	["/bulk4", "https://google.com"],
]);

// Yönlendirmeleri işlemek için middleware
app.use("*", async (c, next) => {
	const path = c.req.path;
	const location = redirectMap.get(path);

	if (location) {
		// Eğer yol yönlendirme haritamızda ise, yönlendirmeyi gerçekleştirin
		return c.redirect(location, 301);
	}

	// Aksi takdirde, bir sonraki handler'a devam edin
	await next();
});

// Herhangi bir yönlendirmeyle eşleşmeyen istekler için varsayılan handler
app.all("*", async (c) => {
	// Origin'e geçirin
	return fetch(c.req.raw);
});

export default app;
```
