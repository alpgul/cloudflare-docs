```js
export default {
	async fetch(request) {
		/**
		 * Farklı URL'leri getirmek için bir nesne
		 * @param {Object} ORIGINS
		 */
		const ORIGINS = {
			"starwarsapi.yourdomain.com": "swapi.dev",
			"google.yourdomain.com": "www.google.com",
		};

		const url = new URL(request.url);

		// Gelen hostname'in ORIGINS nesnesinde bir anahtar olup olmadığını kontrol edin
		if (url.hostname in ORIGINS) {
			const target = ORIGINS[url.hostname];
			url.hostname = target;
			// Eğer öyleyse, isteği o üçüncü taraf origin'e proxy'leyin
			return fetch(url.toString(), request);
		}
		// Aksi takdirde, isteği normal olarak işleyin
		return fetch(request);
	},
};
```

```ts
import { Hono } from "hono";
import { proxy } from "hono/proxy";

// Farklı URL'leri getirmek için bir nesne
const ORIGINS: Record<string, string> = {
	"starwarsapi.yourdomain.com": "swapi.dev",
	"google.yourdomain.com": "www.google.com",
};

const app = new Hono();

app.all("*", async (c) => {
	const url = new URL(c.req.url);

	// Gelen hostname'in ORIGINS nesnesinde bir anahtar olup olmadığını kontrol edin
	if (url.hostname in ORIGINS) {
		const target = ORIGINS[url.hostname];
		url.hostname = target;

		// Eğer öyleyse, isteği o üçüncü taraf origin'e proxy'leyin
		return proxy(url, c.req.raw);
	}

	// Aksi takdirde, isteği normal olarak işleyin
	return proxy(c.req.raw);
});

export default app;
```

from workers import WorkerEntrypoint
from js import fetch, URL

class Default(WorkerEntrypoint):
    async def fetch(self, request):
        # Farklı URL'leri getirmek için bir sözlük
        ORIGINS = {
          "starwarsapi.yourdomain.com": "swapi.dev",
          "google.yourdomain.com": "www.google.com",
        }

        url = URL.new(request.url)

        # Gelen hostname'in ORIGINS nesnesinde bir anahtar olup olmadığını kontrol edin
        if url.hostname in ORIGINS:
            url.hostname = ORIGINS[url.hostname]
            # Eğer öyleyse, isteği o üçüncü taraf origin'e proxy'leyin
            return fetch(url.toString(), request)

        # Aksi takdirde, isteği normal olarak işleyin
        return fetch(request)
