```js
/**
 * HTTP Basic şemasını kullanarak erişimi nasıl kısıtlayacağınızı gösterir.
 * @see https://developer.mozilla.org/en-US/docs/Web/HTTP/Authentication
 * @see https://tools.ietf.org/html/rfc7617
 *
 */

import { Buffer } from "node:buffer";

const encoder = new TextEncoder();

/**
 * `timingSafeEqual` kullanarak değerleri güvenli bir şekilde karşılaştırarak timing saldırılarına karşı koruma sağlar.
 * Daha fazla ayrıntı için https://developers.cloudflare.com/workers/runtime-apis/web-crypto/#timingsafeequal adresine bakın
 * @param {string} a
 * @param {string} b
 * @returns {boolean}
 */
function timingSafeEqual(a, b) {
	const aBytes = encoder.encode(a);
	const bBytes = encoder.encode(b);

	if (aBytes.byteLength !== bBytes.byteLength) {
		// Dizeler crypto.subtle.timingSafeEqual ile karşılaştırılabilmesi için
		// aynı uzunlukta olmalıdır
		return false;
	}

	return crypto.subtle.timingSafeEqual(aBytes, bBytes);
}

export default {
	/**
	 *
	 * @param {Request} request
	 * @param {{PASSWORD: string}} env
	 * @returns
	 */
	async fetch(request, env) {
		const BASIC_USER = "admin";

		// Bir admin şifresine ihtiyacınız olacak. Bu,
		// Worker'ınıza şifrelenmiş gizli anahtar olarak eklenmelidir.
		// Daha fazla bilgi için https://developers.cloudflare.com/workers/configuration/secrets/ adresine bakın
		const BASIC_PASS = env.PASSWORD ?? "password";

		const url = new URL(request.url);

		switch (url.pathname) {
			case "/":
				return new Response("Herkes ana sayfaya erişebilir.");

			case "/logout":
				// HTTP 401 döndürerek "Authorization" başlığını geçersiz kılın.
				// Bu, tarayıcıda hemen kimlik bilgilerini tekrar isteyecek bir popup tetikleyeceği için
				// "WWW-Authenticate" başlığı göndermiyoruz.
				return new Response("Çıkış yapıldı.", { status: 401 });

			case "/admin": {
				// Kimlik doğrulandığında "Authorization" başlığı gönderilir.
				const authorization = request.headers.get("Authorization");
				if (!authorization) {
					return new Response("Giriş yapmanız gerekiyor.", {
						status: 401,
						headers: {
							// Kullanıcıyı kimlik bilgileri için uyarır.
							"WWW-Authenticate": 'Basic realm="my scope", charset="UTF-8"',
						},
					});
				}
				const [scheme, encoded] = authorization.split(" ");

				// Authorization başlığı Basic ile başlamalı, ardından bir boşluk gelmelidir.
				if (!encoded || scheme !== "Basic") {
					return new Response("Bozuk authorization başlığı.", {
						status: 400,
					});
				}

				const credentials = Buffer.from(encoded, "base64").toString();

				// Kullanıcı adı ve şifre ilk iki nokta üst üste ile ayrılır.
				//=> örnek: "kullanıcı_adı:şifre"
				const index = credentials.indexOf(":");
				const user = credentials.substring(0, index);
				const pass = credentials.substring(index + 1);

				if (
					!timingSafeEqual(BASIC_USER, user) ||
					!timingSafeEqual(BASIC_PASS, pass)
				) {
					return new Response("Giriş yapmanız gerekiyor.", {
						status: 401,
						headers: {
							// Kullanıcıyı kimlik bilgileri için uyarır.
							"WWW-Authenticate": 'Basic realm="my scope", charset="UTF-8"',
						},
					});
				}

				return new Response("🎉 Özel erişiminiz var!", {
					status: 200,
					headers: {
						"Cache-Control": "no-store",
					},
				});
			}
		}

		return new Response("Bulunamadı.", { status: 404 });
	},
};
```

```rs
use base64::prelude::*;
use worker::*;

#[event(fetch)]
async fn fetch(req: Request, env: Env, _ctx: Context) -> Result<Response> {
    let basic_user = "admin";
    // Bir admin şifresine ihtiyacınız olacak. Bu,
    // Worker'ınıza şifrelenmiş gizli anahtar olarak eklenmelidir.
    // Daha fazla bilgi için https://developers.cloudflare.com/workers/configuration/secrets/ adresine bakın
    let basic_pass = match env.secret("PASSWORD") {
        Ok(s) => s.to_string(),
        Err(_) => "password".to_string(),
    };
    let url = req.url()?;

    match url.path() {
        "/" => Response::ok("Herkes ana sayfaya erişebilir."),
        // HTTP 401 döndürerek "Authorization" başlığını geçersiz kılın.
        // Bu, tarayıcıda hemen kimlik bilgilerini tekrar isteyecek bir popup tetikleyeceği için
        // "WWW-Authenticate" başlığı göndermiyoruz.
        "/logout" => Response::error("Çıkış yapıldı.", 401),
        "/admin" => {
            // Kimlik doğrulandığında "Authorization" başlığı gönderilir.
            let authorization = req.headers().get("Authorization")?;
            if authorization == None {
                let mut headers = Headers::new();
                // Kullanıcıyı kimlik bilgileri için uyarır.
                headers.set(
                    "WWW-Authenticate",
                    "Basic realm='my scope', charset='UTF-8'",
                )?;
                return Ok(Response::error("Giriş yapmanız gerekiyor.", 401)?.with_headers(headers));
            }
            let authorization = authorization.unwrap();
            let auth: Vec<&str> = authorization.split(" ").collect();
            let scheme = auth[0];
            let encoded = auth[1];

            // Authorization başlığı Basic ile başlamalı, ardından bir boşluk gelmelidir.
            if encoded == "" || scheme != "Basic" {
                return Response::error("Bozuk authorization başlığı.", 400);
            }

            let buff = BASE64_STANDARD.decode(encoded).unwrap();
            let credentials = String::from_utf8_lossy(&buff);
            // Kullanıcı adı ve şifre ilk iki nokta üst üste ile ayrılır.
            //=> örnek: "kullanıcı_adı:şifre"
            let credentials: Vec<&str> = credentials.split(':').collect();
            let user = credentials[0];
            let pass = credentials[1];

            if user != basic_user || pass != basic_pass {
                let mut headers = Headers::new();
                // Kullanıcıyı kimlik bilgileri için uyarır.
                headers.set(
                    "WWW-Authenticate",
                    "Basic realm='my scope', charset='UTF-8'",
                )?;
                return Ok(Response::error("Giriş yapmanız gerekiyor.", 401)?.with_headers(headers));
            }

            let mut headers = Headers::new();
            headers.set("Cache-Control", "no-store")?;
            Ok(Response::ok("🎉 Özel erişiminiz var!")?.with_headers(headers))
        }
        _ => Response::error("Bulunamadı.", 404),
    }
}
```

```ts
/**
 * Hono ile HTTP Basic şemasını kullanarak erişimi nasıl kısıtlayacağınızı gösterir.
 * @see https://developer.mozilla.org/en-US/docs/Web/HTTP/Authentication
 * @see https://tools.ietf.org/html/rfc7617
 */

import { Hono } from "hono";
import { basicAuth } from "hono/basic-auth";

// Ortam arayüzünü tanımlayın
interface Env {
	Bindings: {
		USERNAME: string;
		PASSWORD: string;
	};
}

const app = new Hono<Env>();

// Herkese açık ana sayfa - herkes erişebilir
app.get("/", (c) => {
	return c.text("Herkes ana sayfaya erişebilir.");
});

// Admin rotası - Basic Auth ile korunmuş
app.get(
	"/admin",
	async (c, next) => {
		const auth = basicAuth({
			username: c.env.USERNAME,
			password: c.env.PASSWORD,
		});

		return await auth(c, next);
	},
	(c) => {
		return c.text("🎉 Özel erişiminiz var!", 200, {
			"Cache-Control": "no-store",
		});
	},
);

export default app;
```
