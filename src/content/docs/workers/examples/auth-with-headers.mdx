<Tabs syncKey="workersExamples"> <TabItem label="JavaScript" icon="seti:javascript">

```js
export default {
	async fetch(request) {
		/**
		 * @param {string} PRESHARED_AUTH_HEADER_KEY Anahtar için kontrol edilecek özel başlık
		 * @param {string} PRESHARED_AUTH_HEADER_VALUE Sabit kodlanmış anahtar değeri
		 */
		const PRESHARED_AUTH_HEADER_KEY = "X-Custom-PSK";
		const PRESHARED_AUTH_HEADER_VALUE = "mypresharedkey";
		const psk = request.headers.get(PRESHARED_AUTH_HEADER_KEY);

		if (psk === PRESHARED_AUTH_HEADER_VALUE) {
			// Doğru önceden paylaşılan başlık anahtarı sağlandı. Origin'den isteği getirin.
			return fetch(request);
		}

		// Yanlış anahtar sağlandı. İsteği reddedin.
		return new Response("Üzgünüz, geçersiz bir anahtar sağladınız.", {
			status: 403,
		});
	},
};
```

```py
from workers import WorkerEntrypoint, Response, fetch

class Default(WorkerEntrypoint):
    async def fetch(self, request):
        PRESHARED_AUTH_HEADER_KEY = "X-Custom-PSK"
        PRESHARED_AUTH_HEADER_VALUE = "mypresharedkey"

        psk = request.headers[PRESHARED_AUTH_HEADER_KEY]

        if psk == PRESHARED_AUTH_HEADER_VALUE:
            # Doğru önceden paylaşılan başlık anahtarı sağlandı. Origin'den isteği getirin.
            return fetch(request)

        # Yanlış anahtar sağlandı. İsteği reddedin.
        return Response("Üzgünüz, geçersiz bir anahtar sağladınız.", status=403)
```

</TabItem> <TabItem label="Hono" icon="seti:typescript">

```ts
import { Hono } from 'hono';

const app = new Hono();

// Kimlik doğrulama middleware'i ekleyin
app.use('*', async (c, next) => {
  /**
   * Kimlik doğrulama sabitlerini tanımlayın
   */
  const PRESHARED_AUTH_HEADER_KEY = "X-Custom-PSK";
  const PRESHARED_AUTH_HEADER_VALUE = "mypresharedkey";

  // İstek başlığından önceden paylaşılan anahtarı alın
  const psk = c.req.header(PRESHARED_AUTH_HEADER_KEY);

  if (psk === PRESHARED_AUTH_HEADER_VALUE) {
    // Doğru önceden paylaşılan başlık anahtarı sağlandı. Bir sonraki handler'a devam edin.
    await next();
  } else {
    // Yanlış anahtar sağlandı. İsteği reddedin.
    return c.text("Üzgünüz, geçersiz bir anahtar sağladınız.", 403);
  }
});

// Tüm kimlik doğrulanmış istekleri origin'e geçirerek işleyin
app.all('*', async (c) => {
  return fetch(c.req.raw);
});

export default app;
```
