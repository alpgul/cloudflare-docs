export default {
	async fetch(req) {
		const data =
			req.cf !== undefined
				? req.cf
				: { error: "`cf` nesnesi önizleme içinde mevcut değil." };

		return new Response(JSON.stringify(data, null, 2), {
			headers: {
				"content-type": "application/json;charset=UTF-8",
			},
		});
	},
};

```ts
import { Hono } from "hono";

const app = new Hono();

app.get("*", async (c) => {
	// cf nesnesini almak için ham isteğe erişin
	const req = c.req.raw;

	// cf nesnesinin mevcut olup olmadığını kontrol edin
	const data =
		req.cf !== undefined
			? req.cf
			: { error: "`cf` nesnesi önizleme içinde mevcut değil." };

	// Veriyi 2 boşluk girintisi ile biçimlendirilmiş olarak döndürün
	return c.json(data);
});

export default app;
```

</TabItem> <TabItem label="Python" icon="seti:python">

```py
import json
from workers import Response, WorkerEntrypoint
from js import JSON

class Default(WorkerEntrypoint):
	async def fetch(self, request):
		error = json.dumps({ "error": "`cf` nesnesi önizleme içinde mevcut değil." })
		data = request.cf if request.cf is not None else error
		headers = {"content-type":"application/json"}
		return Response(JSON.stringify(data, None, 2), headers=headers)
```
